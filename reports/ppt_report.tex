\begin{frame}{The Problem}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{When disaster strikes, three agencies must act at once:}
      \begin{itemize}
        \item \hl{Medical} — triage \& casualty care
        \item \hl{Rescue} — search \& extraction
        \item \hl{Logistics} — supplies \& transport
      \end{itemize}
      \vspace{0.6em}
      They share \hl{limited resources} over a 5$\times$5 grid of
      disaster zones, with \hl{uncertain severity} revealed only through
      noisy observations.
      \vspace{0.6em}

      \textbf{Central question:} should agents act \hl{greedily} or
      \hg{coordinate}? We answer it with decision theory under
      uncertainty.
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Why it matters}
        Real responses are decentralised. Selfish optimisation causes
        \hl{congestion} (everyone rushes the same zone) and
        \hl{neglect} (severe zones ignored). We quantify the cost of
        non-cooperation and the gain from principled coordination.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% --------------------------- Slide 3 --------------------------------

      \end{itemize}

% --------------------------- Slide 4 --------------------------------
\begin{frame}{Decision Architecture}
  \textbf{Six interacting techniques across the uncertainty stack:}
  \vspace{0.4em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \hl{MDP + Value Iteration} — single-agent optimal policy
        \item \hl{POMDP + Bayesian belief} — acting under hidden
              severity
        \item \hl{Nash Equilibrium} — strategic stage game
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \hl{Nash Q-Learning} — multi-agent RL coordination
        \item \hl{Shapley Value} — fair credit / resource allocation
        \item \hl{AHP + TOPSIS} — multi-criteria strategy ranking
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \begin{block}{The thread}
    Each agency runs an \hl{MDP} for its own policy, reasons about
    hidden state with a \hl{POMDP belief}, plays a \hl{Nash} stage
    game against the others, learns to \hg{cooperate} via
    \hl{Nash-Q}, splits credit \hl{fairly} via \hl{Shapley}, and we
    rank all strategies objectively with \hl{AHP-TOPSIS}.
  \end{block}
\end{frame}

% --------------------------- Slide 5 --------------------------------
\begin{frame}{Optimal Policy under Full Information (MDP)}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Each agent's resource-dispatch problem is a \hl{Markov Decision
      Process}. We solve it with \hl{Value Iteration}:
      \begin{itemize}
        \item Converges in \hl{$\sim$291 iterations}
              ($\theta = 10^{-6}$, $\gamma = 0.95$)
        \item \hl{Policy Iteration agrees on 100\%} of states —
              a correctness cross-check
      \end{itemize}
      \vspace{0.4em}
      This gives the \hg{benchmark} a perfectly-informed, selfish agent
      could achieve — the baseline coordination must beat.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\linewidth]{c.png}\\
      {\color{muted}\scriptsize T4 — Value-Iteration convergence,
      Medical-agent MDP}
    \end{column}
  \end{columns}
\end{frame}

% --------------------------- Slide 6 --------------------------------
\begin{frame}{Reasoning with Hidden Severity (POMDP)}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Severity is \hl{never observed directly}. Agents maintain a
      \hl{Bayesian belief} over severity levels and update it with each
      noisy sensor reading.
      \begin{itemize}
        \item True level = \hl{3}
        \item Belief mode \hg{locks onto the truth} within
              \hl{$\sim$5 observations}
        \item Posterior confidence approaches \hl{1.0}
      \end{itemize}
      \vspace{0.4em}
      Result: robust decisions \hl{before} the situation is fully known.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\linewidth]{6.png}\\
      {\color{muted}\scriptsize T8 — POMDP belief evolution toward the
      true severity level}
    \end{column}
  \end{columns}
\end{frame}

% --------------------------- Slide 7 --------------------------------
\begin{frame}{The Strategic Tension (Nash Equilibrium)}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      The agencies play a \hl{stage game}: cooperate on a severe zone,
      or pursue private gain.
      \begin{itemize}
        \item Uncoordinated overlap $\rightarrow$ \hl{congestion}
              (diminishing returns)
        \item Joint action on severe zones $\rightarrow$ \hg{synergy}
        \item We select the \hl{Pareto-efficient Nash equilibrium}
              via multi-start iterated best response
      \end{itemize}
      \vspace{0.4em}
      The pure-greedy equilibrium is \hl{socially sub-optimal} — the
      classic coordination dilemma.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\linewidth]{d.png}\\
      {\color{muted}\scriptsize T9 — Stage-game payoffs: Nash vs
      Cooperative vs Competitive}
    \end{column}
  \end{columns}
\end{frame}

% --------------------------- Slide 8 --------------------------------
\begin{frame}{Learning to Cooperate (Nash Q-Learning)}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      Agents learn joint policies with \hl{Nash Q-Learning}
      ($\alpha=0.1$, $\gamma=0.95$, $\varepsilon: 1.0\!\rightarrow\!0.05$).
      \begin{itemize}
        \item \hl{1000} training / \hl{100} evaluation episodes
        \item Cooperation rises from \hl{1.02} to \hg{3.27}
              events per episode
        \item Nash-Q beats Greedy on reward:
              \hl{$t = 11.65$, $p \approx 4{\times}10^{-24}$}
      \end{itemize}
      \vspace{0.3em}
      Coordination is \hg{learned}, not imposed — and the gain is
      statistically decisive.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\linewidth]{8.png}\\
      {\color{muted}\scriptsize T10 — Learning curves: Nash Q-Learning
      vs baselines}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Fair Credit \& Objective Ranking}
  \begin{itemize}
    \item \textbf{Shapley value} splits the coalition's worth fairly:
          grand value \hl{$\approx$33.85}, with \hl{Logistics dominant}
          ($\phi_L\approx$34.2, $\phi_R\approx$0.3, $\phi_M\approx-0.6$);
          the efficiency axiom is verified.
    \item \textbf{AHP + TOPSIS} ranks strategies on 4 criteria —
          Response-Time \& Fairness co-primary (weight \hl{0.375} each),
          \hl{CR = 0.000} (fully consistent).
  \end{itemize}
